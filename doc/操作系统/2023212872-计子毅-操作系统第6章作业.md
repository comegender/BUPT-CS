# 操作系统第6章作业

## 1：老和尚-小和尚挑水

信号量：

```c
int mutex1=1, mutex2=1, count=3, empty=10, full=0;
```

取水：

```c
P(full);
P(count); 
P(mutex2);
V(mutex2);
V(count);
V(empty);
```

向缸中倒水：

```c
P(empty);
P(count);
P(mutex1);
V(mutex1);
P(mutex2);
V(mutex2);
V(full);
V(count);
```

## 2：多类型读-写者问题

初始：

```c
int mutex = 1;        
int A_mutex = 1;      
int B_mutex = 1;      
int C_mutex = 1;      
int data_mutex = 1;   

int countA = 0;            
int countB = 0;            
int countC = 0;            
```

A类用户：

```c
P(A_mutex);            
P(mutex);              
countA++;
if (countA == 1) {     
    P(data_mutex);     
}
V(mutex);
V(A_mutex);

read_data();

P(mutex);
countA--;
if (countA == 0) {     
    V(data_mutex);     
}
V(mutex);
```

B类用户：

```c
P(B_mutex);            
P(mutex);              
countB++;
if (countB == 1) {     
    P(data_mutex);     
}
V(mutex);
V(B_mutex);

read_data();

P(mutex);
countB--;
if (countB == 0) {     
    V(data_mutex);     
}
V(mutex);
```

C类用户：

```c
P(C_mutex);            
P(data_mutex);     

write_data();
  
V(data_mutex);     
V(C_mutex)
```

## 3：牙科门诊

初始：

```c
doctor=3, patient=0, mutex1=1,mutex2=1,mutexn=1;
int DState[3] = {0};
int PState[20] = {0};
int n = 0;
```

医生：

```c
int i; 
P(patient);
P(doctor);
P(mutex1);
DState[i] = 1; 
V(mutex1);

work();

P(mutex1);
DState[i] = 0;
V(mutex1);
```

患者：

```c
if (n < 20) {
    int i = n;
    P(mutexn);
    n++;
    V(mutexn);
    V(patient);
    P(mutex2);
    PState[i] = 1;
    V(mutex2);
    
    look();
    
    V(doctor);
    P(mutex2);
    PState[i] = 0;
    V(mutex2);
}
```
